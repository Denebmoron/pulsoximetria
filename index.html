function handleData(event) { const value = new TextDecoder().decode(event.target.value); // El nuevo formato que enviaremos desde la ESP32 será: "BPM,IR,SpO2\n" const dataParts = value.split(','); if (dataParts.length >= 3) { const bpm = parseInt(dataParts[0]); const irValue = parseInt(dataParts[1]); const spo2 = parseInt(dataParts[2]); // Actualizar BPM if(bpm > 0) bpmDisplay.innerText = bpm; // Actualizar SpO2 en el anillo cyan if(spo2 > 0) { document.getElementById('spo2Value').innerText = spo2 + " % SpO2"; } // Dibujar gráfica tipo osciloscopio drawGraph(irValue); } // Modo de compatibilidad si la ESP32 sigue con el código viejo else if (dataParts.length === 2) { const bpm = parseInt(dataParts[0]); const irValue = parseInt(dataParts[1]); if(bpm > 0) bpmDisplay.innerText = bpm; drawGraph(irValue); } }
